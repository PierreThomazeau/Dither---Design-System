## Usage

```tsx
import { StatCard } from "@/components/blocks/stat-card";

<StatCard label="Revenue" value="€482.9k" delta="+12.4%" caption="vs last month" trend="up" sparkline={[12, 14, 13, 18, 21]} />
```
